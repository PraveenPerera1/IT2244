(06)Check which number is greater ans less

echo "Enter two numbers:"
read num1;
read num2;

if [[ "$num1" -gt "$num2" ]]; then
	echo "$num1 is greater than $num2"
fi


if [[ "$num1" -st "$num2" ]]; then
	echo "$num1 is smaller than $num2"
fi

(07)Create simple calculator using "expr" command same as the Question03

echo "Enter the first number:"
read num1                          # Read first number

echo "Enter the second number:"
read num2                          # Read second number

# Use expr for arithmetic operations
sum=$(expr $num1 + $num2)                       
sub=$(expr $num1 - $num2)  
div=$(expr $num1 / $num2)  
mul=$(expr $num1 * $num2)  


echo "Summation:$sum"
echo "Subtraction:$sub"
echo "Division:$div"
echo "Multiplication:$mul"
